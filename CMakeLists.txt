cmake_minimum_required(VERSION 3.20)
project(TTIMU VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(ttimu src/IMULibrary.cpp src/ttimu/detail/InavCore.cpp src/ttimu/detail/InavMath.cpp src/ttimu/detail/InavStartup.cpp)
target_include_directories(ttimu PUBLIC src)
target_compile_features(ttimu PUBLIC cxx_std_11)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(ttimu PRIVATE -Wall -Wextra -Wshadow -ffp-contract=off)
endif()
option(TTIMU_BUILD_TESTS "Build host tests" ON)
if(TTIMU_BUILD_TESTS)
  enable_testing()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME viewer_http COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_viewer_server.py)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/make_reference.py
    --output ${CMAKE_CURRENT_BINARY_DIR}/inav_reference.cpp COMMAND_ERROR_IS_FATAL ANY)
  add_executable(ttimu_tests tests/test_main.cpp ${CMAKE_CURRENT_BINARY_DIR}/inav_reference.cpp)
  target_include_directories(ttimu_tests PRIVATE tests)
  target_link_libraries(ttimu_tests PRIVATE ttimu)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(ttimu_tests PRIVATE -ffp-contract=off)
  endif()
  add_test(NAME ttimu_host COMMAND ttimu_tests)
  add_executable(ttimu_header_compat tests/test_header_compat.cpp)
  target_link_libraries(ttimu_header_compat PRIVATE ttimu)
  add_test(NAME ttimu_header_compat COMMAND ttimu_header_compat)
  add_executable(ttimu_startup_tests tests/test_startup.cpp)
  target_link_libraries(ttimu_startup_tests PRIVATE ttimu)
  add_test(NAME ttimu_startup COMMAND ttimu_startup_tests)
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME viewer_parser COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/viewer.test.mjs)
  endif()
endif()
